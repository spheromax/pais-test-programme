\section{Stress, strength and stored energy}

\subsection{The trace of the Maxwell stress tensor}
\meta{[D] run 3 $\cdot$ foundation for I.2--I.7}
\begin{statement}
$T_{ij}=\varepsilon_0\big(E_iE_j-\tfrac12\delta_{ij}E^2\big)+\mu_0^{-1}\big(B_iB_j-\tfrac12\delta_{ij}B^2\big)$ and
\begin{equation}
\tr T = -\Big(\frac{\varepsilon_0E^2}{2}+\frac{B^2}{2\mu_0}\Big) = -u .\label{eq:trT}
\end{equation}
In statics the force density on charges and currents is $f_i=\rho E_i+(\mathbf J\times\mathbf B)_i=\partial_jT_{ij}$.
\end{statement}
\begin{proof}
With $\delta_{ii}=3$: $\varepsilon_0(E^2-\tfrac32E^2)=-\tfrac12\varepsilon_0E^2$, and the same for $B$. The force identity is Maxwell's
equations with $\partial_t(\varepsilon_0\mathbf E\times\mathbf B)=0$ \cite{jackson}.
\end{proof}

\subsection{Virial theorem for a static, self-contained field: von Laue's theorem}
\meta{[P] run 3 $\cdot$ gaps 2, 4, 6 $\cdot$ leads to I.3, I.4, I.7, III.1}
\begin{theorem}\label{thm:virial}
Let a body of finite extent be in static equilibrium under its own electromagnetic fields (no external force), with mechanical
stress $\sigma_{ij}$, and let the fields far away fall off at least as $E,B=O(r^{-2})$ (net charge allowed). Then
\begin{equation}
\int_{\rm body}\tr\sigma\,dV \;=\; \int_{\rm all\ space}u\,dV \;=\; U .\label{eq:virial}
\end{equation}
Holding a field energy $U$ requires a net integrated tension equal to $U$, whatever the geometry.
\end{theorem}
\begin{proof}
Inside the body $\partial_j\sigma_{ij}+f_i=0$ with $f_i=\partial_jT_{ij}$. Put $\Sigma_{ij}=\sigma_{ij}+T_{ij}$ ($\sigma=0$ outside). Then
$\partial_j\Sigma_{ij}=0$ everywhere, including across the surface, where the jump of the mechanical traction balances the jump of
$T_{ij}n_j$ (the electromagnetic surface force). For a ball $B_R$ containing the body,
\[
0=\int_{B_R}x_i\,\partial_j\Sigma_{ij}\,dV=\oint_{\partial B_R}x_iT_{ij}n_j\,dS-\int_{B_R}\Sigma_{ii}\,dV .
\]
On $\partial B_R$, $|T|\le u=O(R^{-4})$, so the surface term is $O(R\cdot R^{-4}\cdot R^2)=O(R^{-1})\to0$. Hence
$\int\tr\sigma\,dV=-\int\tr T\,dV=U$ by \eqref{eq:trT}.
\end{proof}
\noindent\emph{Remarks.} (i) The same argument applied to the full relativistic stress--energy tensor of a static closed system gives
$\int T^{ij}\,d^3x=0$ (von Laue, 1911 \cite{laue}); \eqref{eq:virial} is its trace. It is why such a system's energy and momentum form a
four-vector with mass $E/c^2$: internal stresses produce no inertial anomaly. (ii) Configurations without stress
(``force-free'' fields) exist only with infinite extent \cite{hilal,cr5}.
\refs{\cite{jackson,laue,moon,hilal,tsutsui,lbl}}

\subsection{Minimum structural mass; no force-free finite magnet}
\meta{[P] run 3 $\cdot$ gaps 6 and B1 $\cdot$ from I.2, leads to I.5, I.6}
\begin{corollary}\label{cor:mass}
If every principal stress obeys $\sigma_k\le\sigma_w$, then $\tr\sigma\le3\sigma_w$ and $M\ge\rho U/(3\sigma_w)$. When the load is carried by
uniaxial members (fibres, hoops, tapes), $\tr\sigma=\sigma_1\le\sigma_w$ and
\begin{equation}
M\;\ge\;\frac{\rho\,U}{\sigma_w},\qquad\text{i.e.}\qquad \frac{U}{M}\le\frac{\sigma_w}{\rho}.\label{eq:mmin}
\end{equation}
Compression members only add mass. If $\sigma\equiv0$ then $U=0$: a finite body cannot hold field energy without stress.
\end{corollary}
\begin{proof}Integrate the pointwise bounds in \eqref{eq:virial} over the stressed volume.\end{proof}

\subsection{Spinning bodies: the virial theorem for rotation, and the rim speed}
\meta{[P] run 3 (ring speed R0) $\cdot$ gap 2 $\cdot$ from I.2, leads to I.7}
\begin{theorem}\label{thm:spin}
For a body rotating steadily at $\Omega$, free of external forces and surface tractions, $\displaystyle\int\tr\sigma\,dV=2T$ with
$T=\tfrac12\int\rho\Omega^2r_\perp^2\,dV$.
\end{theorem}
\begin{proof}
$\partial_j\sigma_{ij}=\rho a_i$ with $\mathbf a=-\Omega^2\mathbf r_\perp$. Multiply by $x_i$ and integrate; the traction-free boundary
kills the surface term: $-\int\tr\sigma\,dV=\int\rho\,\mathbf x\cdot\mathbf a\,dV=-\int\rho\Omega^2r_\perp^2\,dV=-2T$.
\end{proof}
\begin{corollary}
$T/M\le\tfrac12\max(\tr\sigma)/\rho$. A uniaxial hoop gives $T/M\le\sigma/(2\rho)$ (shape factor $K=\tfrac12$); a biaxial constant-stress disk
gives $K=1$: the flywheel law $E/M=K\sigma/\rho$ with $K\le1$ in plane stress. For a thin ring, an arc $d\theta$ of mass
$\rho AR\,d\theta$ needs centripetal force $\rho A v^2d\theta$, supplied by $2\sigma A\sin(d\theta/2)\approx\sigma A\,d\theta$, so
\begin{equation}
\sigma=\rho v^2,\qquad v_{\max}=\sqrt{\sigma/\rho}.\label{eq:ring}
\end{equation}
\end{corollary}
\nums{CT-11 rim, $R=2\un{m}$ at 30\,000 rpm: $v=6.28\un{km/s}$, needing $\sigma/\rho=3.95\e7\un{J/kg}$, i.e.\ 51 GPa at $1300\un{kg/m^3}$.
Ring limits are in the table of I.3. Measured: composite rotors reached $965\un{m/s}$ tip speed and $>310\un{kJ/kg}$ \cite{sae}.}

\subsection{Magnet windings: pressure, hoop stress and the field ceiling}
\meta{[D] run 0, corrected in run 3 $\cdot$ gap 6 $\cdot$ from I.1, I.3; leads to I.6}
\begin{statement}
A winding with field $B$ inside and $\approx0$ outside feels the magnetic pressure $p=B^2/2\mu_0$ (the jump of $T_{nn}$). A thin solenoid of
radius $R$ and wall $t$ carries the hoop stress
\begin{equation}
\sigma_\theta=\frac{pR}{t}=\frac{B^2R}{2\mu_0t},\qquad\text{and for a thick winding }(t\approx R)\qquad B_{\max}\approx\sqrt{2\mu_0\sigma_w}.\label{eq:bmax}
\end{equation}
\end{statement}

the pressure times $R/t$.

\subsection{The specific-strength ceiling and the field-mass bound}
\meta{[P] run 3 $\cdot$ gaps 2, 4, 6 $\cdot$ from I.2--I.4; leads to III.1}
\begin{theorem}\label{thm:fieldmass}
For any material-supported static field (Theorem~\ref{thm:virial}) or spinning body (Theorem~\ref{thm:spin}),
$U/M\le3\sigma_w/\rho$ and $T/M\le\tfrac32\sigma_w/\rho$. The mass-energy added is positive and bounded:
\begin{equation}
0<\frac{\Delta m}{M}=\frac{U}{Mc^2}\le\frac{3\sigma_w}{\rho c^2}.\label{eq:dm}
\end{equation}
\end{theorem}
\begin{proof}Theorem~\ref{thm:virial} or \ref{thm:spin} with Corollary~\ref{cor:mass}, then $\Delta m=U/c^2$.\end{proof}
\nums{The largest $\sigma/\rho$ of any known material is graphene's ideal strength, $130\un{GPa}/2200\un{kg\,m^{-3}}=5.9\e7\un{J/kg}$
($v=7.7\un{km/s}$). That is the order of chemical energy per kilogram (TNT $4.2\e6$, hydrogen--oxygen $1.3\e7\un{J/kg}$), as it must be:
both are bond energies. So $\Delta m/M\le2.0\e{-9}$ (triaxial) or $6.6\e{-10}$ (uniaxial). The run-2 rig stores
$U=Q^2/2C=0.047\un{J}$ ($C=8\varepsilon_0R$), so $\Delta m=+5.2\e{-19}\un{kg}$.}
\verdict{Stored field energy or spin can never reduce inertial mass, and can raise it by at most about $10^{-9}$ of the structure. The
``mass reduction by field energy'' route of gap 4 is cut by the materials wall plus $E=mc^2$.}

\subsection{Weakest link: Weibull size scaling}
\meta{[D] run 3 $\cdot$ gap 2 $\cdot$ leads to I.9, I.10}
\begin{statement}
If a fibre of length $L$ is a chain of $L/L_0$ independent links, each surviving stress $\sigma$ with probability
$\exp[-(\sigma/\sigma_0)^m]$, then
\begin{equation}
S_L(\sigma)=\exp\!\Big[-\frac{L}{L_0}\Big(\frac{\sigma}{\sigma_0}\Big)^m\Big],\quad
\bar\sigma(L)=\sigma_0\Big(\frac{L_0}{L}\Big)^{1/m}\Gamma\!\Big(1+\frac1m\Big),\quad
\frac{\sigma(L)}{\sigma(L_{\rm ref})}=\Big(\frac{L_{\rm ref}}{L}\Big)^{1/m}.\label{eq:weibull}
\end{equation}
\end{statement}
\begin{proof}Independent links: the survival probabilities multiply. The mean follows from $\int_0^\infty S_L\,d\sigma$.\end{proof}
\nums{Hoop circumference $2\pi\cdot2\un{m}=12.57\un{m}$. CNT fibre, 8.2 GPa at 10 mm $\to$ 2.0 / 4.0 / 5.7 GPa at 12.57 m for $m=5/10/20$.
Bundles, 80 GPa at 2 cm $\to$ 22 / 42 / 58 GPa. Meeting the 51.3 GPa requirement needs $m\ge14.5$ (run-7 correction: on the wall area
the requirement is 89 GPa, above the 80 GPa measured at 2 cm, so no $m$ suffices; part 5, note on I.11). Measured: CNT fibre strength fell to
about 1 GPa as the gauge length rose from 1 to 10 mm \cite{ma}; mean and spread fall with gauge length \cite{beyerlein}; intrinsic
$\sim$120 GPa against fibres below 10 GPa \cite{gao}.}
\verdict{Error logged in run 3: runs 1--2 quoted the cm-gauge 80 GPa as if it held at metre scale.}

\subsection{Equal load sharing: Daniels' bundle theorem}
\meta{[D] run 3 $\cdot$ gap 2 $\cdot$ from I.8, leads to I.10}
\begin{statement}
For $N$ fibres with independent strengths (distribution $F$) sharing load equally, the strength per fibre tends, as $N\to\infty$, to
\begin{equation}
\sigma^*=\max_x\,x\,[1-F(x)];\qquad\text{for Weibull fibres}\quad \sigma^*=\sigma_0(me)^{-1/m}.\label{eq:daniels}
\end{equation}
\end{statement}
\begin{proof}[Sketch]
At load $s$ per original fibre, the survivors (fraction $1-F(x)$) each carry $x$ with $s=x[1-F(x)]$. An equilibrium exists only while $s$ is
below the maximum of the right-hand side; beyond it the bundle fails. For Weibull $F$, $\tfrac{d}{dx}\big(xe^{-(x/\sigma_0)^m}\big)=0$ gives
$(x/\sigma_0)^m=1/m$ \cite{daniels}.
\end{proof}
\nums{Bundle strength over mean fibre strength, $(me)^{-1/m}/\Gamma(1+1/m)$: 0.65, 0.76, 0.84 for $m=5,10,20$ (the ``Daniels effect'' in
nanotube bundles \cite{bai}).}

\subsection{Composites: the chain of bundles, and what gap 2 now needs}
\meta{[D]+[E] run 3 $\cdot$ gap 2 $\cdot$ from I.8, I.9}
\begin{statement}
A unidirectional composite of length $L$ fails as a chain of $L/\delta$ bundles, each of the ineffective length $\delta$ over which the matrix
reloads a broken fibre. Each bundle has the Weibull scale $\sigma_0(L_0/\delta)^{1/m}$ and Daniels strength, and the chain is only weakly
length-dependent, so
\begin{equation}
\sigma_c\approx V_f\,\sigma_0\Big(\frac{L_0}{\delta\,m\,e}\Big)^{1/m}.\label{eq:rosen}
\end{equation}
The length that matters is $\delta$, set by load transfer, not $L$ \cite{rosen}.
\end{statement}
\nums{With $L_0=10\un{mm}$: $\delta=0.1\un{mm}$ gives factors 1.49 / 1.14 / 1.03 for $m=5/10/20$; $\delta=1\un{mm}$ gives 0.94 / 0.91 / 0.92.
Gap 2 restated: a $V_f=0.6$ composite of 80 GPa bundles ($\rho\approx1260$) gives at most 48 GPa against 49.7 GPa needed (0.97$\times$).
Run-7 correction: with the wall-area density the bundles themselves reach only 0.90$\times$ (part 5, note on I.11).}
\verdict{Gap 2 sits at the edge of the best cm-scale material. The in-between R\&D gap is metre-long ultralong-nanotube bundles in a matrix
with $\delta\lesssim0.1\un{mm}$ and $V_f\ge0.62$. Kill test: measure bundle strength at 2, 20 and 200 mm gauge; if $m<10$, the nanotube
route to the rim is cut. Today's buildable best, a carbon-fibre hoop, is $4.0\times$ short in speed.}

\section{Radiation from charged, moving bodies}

\subsection{The Pais flux formula as filed}
\meta{[D] run 0 $\cdot$ canary in run 3}
\begin{statement}
Eq.~1 of US 10,135,366: $S_{\max}=f_G\gamma(\sigma^2/\varepsilon_0)[R_r\omega+R_v\nu+v_R]$; Eq.~2 replaces the bracket by $R_v\nu^2t_{\rm op}$.
Units: $(\mathrm{J\,m^{-3}})(\mathrm{m\,s^{-1}})=\mathrm{W\,m^{-2}}$. It is an energy density carried at a surface speed, not radiation.
\end{statement}
\nums{Sphere, $f_G=\tfrac23$, $\sigma=5\e4\un{C/m^2}$, $R=2\un{m}$, 30\,000 rpm: $1.18\e{24}\un{W/m^2}$ (canary re-derived: 0.2 \%).
In Eq.~2 the ``speed'' is acceleration $\times$ time: for the run-2 rig (80 nm, 100 MHz, $t_{\rm op}=1\un{s}$) it is $3.2\e{10}\un{m/s}=105\,c$.}

\subsection{Stationary sources do not radiate}
\meta{[P] run 0 $\cdot$ CT-11 verdict $\cdot$ leads to II.3, II.5}
\begin{theorem}\label{thm:static}
If $\rho(\mathbf x)$ and $\mathbf J(\mathbf x)$ are time-independent and localized, no power crosses a sphere at infinity.
\end{theorem}
\begin{proof}
The fields are static: $E=O(r^{-2})$, and a localized steady current has no magnetic monopole term, so $B=O(r^{-3})$. Then
$|\mathbf S|=|\mathbf E\times\mathbf B|/\mu_0=O(r^{-5})$ and $\oint\mathbf S\cdot d\mathbf A=O(r^{-3})\to0$.
\end{proof}
\noindent A uniformly spinning axisymmetric charged body has time-independent $\rho$ and $\mathbf J$: it radiates 0 W.

\subsection{A breathing sphere does not radiate}
\meta{[P] run 0 $\cdot$ from II.2}
\begin{proof}[Statement and proof]
For spherically symmetric $\rho(r,t)$ and radial $\mathbf J(r,t)$, $\mathbf B$ would be a spherically symmetric axial field, hence zero (no
monopoles). Outside the charge Gauss gives $\mathbf E=Q\hat{\mathbf r}/(4\pi\varepsilon_0r^2)$ with $Q$ fixed: static. Inside,
$\varepsilon_0\partial_t\mathbf E=-\mathbf J$. Nothing radiates (the electromagnetic analogue of Birkhoff's theorem).
\end{proof}

\subsection{Larmor: oscillating and rotating dipoles}
\meta{[D] $\cdot$ used in II.5--II.8}
\begin{statement}
A dipole $\mathbf p(t)$ radiates $P=|\ddot{\mathbf p}|^2/(6\pi\varepsilon_0c^3)$. Hence
\begin{equation}
\langle P\rangle_{\rm osc}=\frac{p_0^2\omega^4}{12\pi\varepsilon_0c^3}\quad(\mathbf p=p_0\cos\omega t\,\hat{\mathbf z}),\qquad
P_{\rm rot}=\frac{p_0^2\omega^4}{6\pi\varepsilon_0c^3}\quad(\text{fixed }p_0\text{ rotating in a plane}).\label{eq:larmor}
\end{equation}
\end{statement}
\noindent For the rotating dipole $|\ddot{\mathbf p}|=p_0\omega^2$ at all times; for the oscillating one $\langle\cos^2\rangle=\tfrac12$. Check: the multipole
formula \eqref{eq:multipole} with $l=1$ reproduces \eqref{eq:larmor} exactly (numerical ratio 1.000).

\subsection{Selection rule for $C_n$-symmetric spinning charge (H1, the triangle)}
\meta{[P] run 3 $\cdot$ hypothesis H1 $\cdot$ from II.2, II.4}
\begin{theorem}\label{thm:cn}
Let $\rho_0(\mathbf x)$ be invariant under rotation by $2\pi/n$ about $z$, and let $\rho(\mathbf x,t)=\rho_0(R_{-\Omega t}\mathbf x)$. Then the
body radiates only at the harmonics $kn\Omega$ ($k=1,2,\dots$), the harmonic $kn\Omega$ comes only from multipoles $l\ge kn$, and the lowest
radiating multipole is $l=n$ at frequency $n\Omega$. As $n\to\infty$ (axisymmetry) nothing radiates.
\end{theorem}
\begin{proof}
Expand $\rho_0=\sum_m\rho_m(r,\theta)e^{im\phi}$. Invariance under $\phi\to\phi+2\pi/n$ forces $e^{2\pi im/n}=1$, so $m\in n\mathbb Z$. Rotation
replaces $\phi$ by $\phi-\Omega t$, so the $m$ term oscillates at $|m|\Omega$ and only $m=0$ is static (Theorem~\ref{thm:static}). Its multipole
moments $Q_{lm'}\propto\int e^{im\phi}e^{-im'\phi}d\phi=2\pi\delta_{mm'}$ vanish unless $m'=m$, and $Y_{lm'}$ exists only for $|m'|\le l$. The
magnetic multipoles, from $\mathbf J=\rho\,\Omega\hat{\mathbf z}\times\mathbf x$, obey the same rule.
\end{proof}
\noindent In the long-wavelength limit the electric $(l,m)$ power is (Jackson 9.155 with 9.169 \cite{jackson})
\begin{equation}
P_E(l,m)=\frac{c}{2\varepsilon_0}\,\frac{l+1}{l}\,\frac{k^{2l+2}\,|Q_{lm}|^2}{[(2l+1)!!]^2},\qquad k=\frac{|m|\Omega}{c},\label{eq:multipole}
\end{equation}
with $Q_{lm}$ the complex amplitude of the $e^{-i|m|\Omega t}$ component (twice the $m>0$ Fourier coefficient).
\inputs{charges $Q/3$ at the vertices of an equilateral triangle of circumradius $a=1\un{m}$; $Q=1\un{mC}$ (assumed, near vacuum
breakdown for a 1 m body); $\Omega=320\un{rad/s}$, the 6061 rim limit $\sqrt{276\un{MPa}/2700}$.}
\nums{Dipole and quadrupole vanish identically; first is $l=m=3$ at $3\Omega$, with
\[
Q_{33}=2\sum_{k=0}^{2}\frac Q3\,a^3\,Y^*_{33}\big(\tfrac\pi2,\phi_k\big)=-\frac14\sqrt{\frac{35}{\pi}}\,Qa^3,\qquad |Q_{33}|=0.834\,Qa^3 .
\]
Then $ka=3.2\e{-6}$ and
$P=1.6\e{-35}\un{W}$. Since $P\propto\Omega^8$, 1 W needs $\Omega=7.2\e6\un{rad/s}$, a rim speed of $0.024\,c$, 930 times the ideal-graphene
limit (I.7). A 0.1 \% vertex-charge imbalance leaves a rotating dipole $p=3.3\e{-7}\un{C\,m}$ radiating $2.6\e{-19}\un{W}$:
$10^{16}$ times the octupole, and still nothing.}
\verdict{H1 cut: the symmetry that makes the triangle elegant suppresses its radiation as $(\Omega a/c)^8$, and the speed needed crosses the
materials wall.}

\subsection{Rotating in-plane ferroelectric (H2)}
\meta{[E] run 3 $\cdot$ hypothesis H2 $\cdot$ from II.4, I.4}
\inputs{BiFeO$_3$ disk, $R=5\un{cm}$, $t=1\un{mm}$; $P_s=1.0\un{C/m^2}$ (bulk; strained films reach 1.3--1.5); strength 100 MPa
(assumed ceramic); $\rho=8340$.}
\nums{$p_0=P_sV=7.9\e{-6}\un{C\,m}$. At $\Omega=10^4\un{rad/s}$, \eqref{eq:larmor} gives $1.4\e{-10}\un{W}$ ($3.1\e{-10}$ with the film value).
But the rim would move at 500 m/s while the disk bursts at $\sqrt{\sigma/\rho}=110\un{m/s}$, so $\Omega\le2.2\e3\un{rad/s}$ and
$P\le3.2\e{-13}\un{W}$. Worse, the uncompensated in-plane polarization of a thin disk makes a depolarizing field $NP_s/\varepsilon_0$ with
$N\approx\tfrac\pi4\,t/2R=7.9\e{-3}$, i.e.\ $8.9\e8\un{V/m}$, far above coercive fields ($\sim10^7\un{V/m}$): the disk breaks into domains
and the net moment collapses.}
\verdict{H2 cut ($\omega^4$ smallness, burst speed, depolarization).}

\subsection{The decisive rig: known physics against Pais Eq.~2}
\meta{[D] run 2 design $\cdot$ canary in run 3 $\cdot$ gap 4}
\inputs{30 cm aluminium disk ($R=0.15\un{m}$), $Q=1\,\mu$C, rigid vibration $\pm80\un{nm}$ at 100 MHz, $t_{\rm op}=1\un{s}$, $f_G=1$ (``equal to 1

\nums{Known physics: $p_0=Qx_0=8.0\e{-14}\un{C\,m}$, so $P=1.1\e{-7}\un{W}$. $C=8\varepsilon_0R=1.06\e{-11}\un{F}$, $V=94\un{kV}$,

\subsection{Eq.~2 has already been tested: capacitive ultrasound transducers}
\meta{[E] run 3 $\cdot$ in-between finding $\cdot$ gap 4}
\begin{statement}
A capacitive micromachined ultrasonic transducer (CMUT) is a metallized membrane over a 0.05--3 $\mu$m gap, held at a DC field of order
$10^8\un{V/m}$ and driven at 10 kHz--60 MHz \cite{cmut,ergun}: a charged plate vibrating at megahertz, which is every ingredient of Eq.~2.
\end{statement}
\inputs{$E=10^8\un{V/m}$, so $\sigma=\varepsilon_0E=8.9\e{-4}\un{C/m^2}$ (125 times the rig); 5 MHz; amplitude 10 nm (assumed; at most a third of

\nums{Eq.~2: $S=8.7\e{11}\un{W/m^2}$, i.e.\ $8.7\e7\un{W}$ from every square centimetre of array after one second and $3.1\e{11}\un{W}$ after an
hour. Known physics: the cell charge ($8.9\e{-8}\un{C}$ per cm$^2$) moving 10 nm radiates $8.5\e{-17}\un{W}$. Ratio $10^{24}$.}
\verdict{Hand-held CMUT probes run on a few watts. Eq.~2 is falsified by devices in daily use; the decisive test of the flux formula has
already been run, many times over. The premise of gap 4 (that Eq.~2's flux exists) fails.}
